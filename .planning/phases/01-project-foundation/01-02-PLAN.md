  Create `backend/src/middleware/errorHandler.js`:
  ```javascript
  const { NODE_ENV } = require('../config/env');

  const notFound = (req, res, next) => {
    const error = new Error(`Not Found - ${req.originalUrl}`);
    res.status(404);
    next(error);
  };

  const errorHandler = (err, req, res, next) => {
    const statusCode = res.statusCode === 200 ? 500 : res.statusCode;
    res.status(statusCode).json({
      success: false,
      message: err.message,
      stack: NODE_ENV === 'production' ? undefined : err.stack,
    });
  };

  module.exports = { notFound, errorHandler };
  ```

  Key decisions:
  - Consistent error response format: `{ success: false, message, stack? }`
  - Stack trace only in development (hide in production)
  - 404 handler as separate middleware (catches unmatched routes)
  - Status code passthrough (if already set by upstream middleware)

  <action>
  Create `backend/src/server.js`:
  ```javascript
  const express = require('express');
  const cors = require('cors');
  const helmet = require('helmet');
  const morgan = require('morgan');
  const connectDB = require('./config/db');
  const { PORT, NODE_ENV, CORS_ORIGIN } = require('./config/env');
  const apiRoutes = require('./routes');
  const { notFound, errorHandler } = require('./middleware/errorHandler');

  const app = express();

  // Security middleware
  app.use(helmet());

  // CORS configuration
  app.use(cors({
    origin: CORS_ORIGIN,
    credentials: true,
  }));

  // Request logging
  app.use(morgan(NODE_ENV === 'production' ? 'combined' : 'dev'));

  // Body parsing
  app.use(express.json({ limit: '10mb' }));
  app.use(express.urlencoded({ extended: true }));

  // API routes — all under /api/v1/ prefix
  app.use('/api/v1', apiRoutes);

  // Root route
  app.get('/', (req, res) => {
    res.json({
      success: true,
      message: 'InferaNews Backend API',
      version: 'v1',
      docs: '/api/v1/health',
    });
  });

  // Error handling
  app.use(notFound);
  app.use(errorHandler);

  // Start server
  const startServer = async () => {
    await connectDB();
    app.listen(PORT, () => {
      console.log(`Server running in ${NODE_ENV} mode on port ${PORT}`);
    });
  };

  startServer();

  module.exports = app;
  ```

  Key decisions:
  - Middleware order: helmet → cors → morgan → body parsers → routes → error handlers
  - Morgan uses 'dev' format in development (colored, concise) and 'combined' in production (Apache-style)
  - JSON body limit set to 10mb (articles may contain long content)
  - CORS uses the configured origin from env
  - Server starts AFTER DB connection (fail-fast)
  - App exported for potential testing use
  - Root route returns API info and version
  - All API routes under `/api/v1` prefix (SETUP-04)
